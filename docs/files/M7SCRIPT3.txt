
local url = ""

local items = {
    { label = "Apples",   value = 12, color = "#ef476f" },
    { label = "Bananas",  value = 19, color = "#ffd166" },
    { label = "Cherries", value = 7,  color = "#06d6a0" },
    { label = "Dates",    value = 15, color = "#118ab2" },
}

local htmlTemplate = [=[
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE html>
<html xmlns="http://www.w3.org/1999/xhtml">
<head>
    <meta charset="UTF-8" />
    <title>Pie Chart</title>
    <style>
    .pie-wrapper { display: flex; align-items: center; gap: 24px; font-family: sans-serif; }
    .pie { width: 720px; height: 720px; border-radius: 50%; }
    .legend { list-style: none; padding: 0; margin: 0; font-size: 24px; }
    .legend li { display: flex; align-items: center; gap: 24px; margin-bottom: 18px; }
    .dot { width: 36px; height: 36px; border-radius: 9px; display: inline-block; }
    </style>
</head>
<body>
    <div class="pie-wrapper">
      <div id="pie" class="pie"></div>
      <ul id="legend" class="legend"></ul>
    </div>

    <script>
    //<![CDATA[
      const items = @ITEMS@;
    
      const total = items.reduce((sum, item) => sum + item.value, 0) || 1;
      let currentPct = 0;
    
      const stops = items.map(item => {
        const start = currentPct;
        currentPct += (item.value / total) * 100;
        return `${item.color} ${start}% ${currentPct}%`;
      });
    
      document.getElementById('pie').style.background = `conic-gradient(${stops.join(', ')})`;
    
      const legend = document.getElementById('legend');
      items.forEach(item => {
        const li = document.createElement('li');
        const dot = document.createElement('span');
        dot.className = 'dot';
        dot.style.background = item.color;
    
        li.append(dot, ` ${item.label}: ${item.value}`);
        legend.appendChild(li);
      });
    //]]>
    </script>
</body>
</html>
]=]

local function show(url)
    ll.SetPrimMediaParams(2, {
        PRIM_MEDIA_CURRENT_URL, url,
        PRIM_MEDIA_HOME_URL, url,
        PRIM_MEDIA_AUTO_ZOOM, false,
        PRIM_MEDIA_FIRST_CLICK_INTERACT, true,
        PRIM_MEDIA_PERMS_INTERACT, PRIM_MEDIA_PERM_ANYONE,
        PRIM_MEDIA_PERMS_CONTROL, PRIM_MEDIA_PERM_NONE,
        PRIM_MEDIA_AUTO_PLAY, true,
        PRIM_MEDIA_WIDTH_PIXELS, 1024,
        PRIM_MEDIA_HEIGHT_PIXELS, 1024,
    })
end

function LLEvents.http_request(id, method, body)
    if method == URL_REQUEST_GRANTED then
        url = body
        print(url)
        show(url)
    elseif method == URL_REQUEST_DENIED then
        print("Unable to get URL!")
    elseif method == "GET" then
        local html = string.gsub(htmlTemplate, "@ITEMS@", function()
            return lljson.encode(items)
        end)
        ll.SetContentType(id, CONTENT_TYPE_XHTML)
        ll.HTTPResponse(id, 200, html)
    end
end

function LLEvents.on_rez(start_param)
    ll.ResetScript()
end

function LLEvents.changed(change)
    if bit32.btest(change, bit32.bor(CHANGED_REGION_START, CHANGED_OWNER, CHANGED_INVENTORY)) then
        ll.ResetScript()
    end
end
