const container = document.getElementById("content");
const statusEl = document.getElementById("status");
const unreadBadge = document.getElementById("unreadBadge");
const unreadCountEl = document.getElementById("unreadCount");
const POLL_INTERVAL = 15000; // 15 seconds
let pollTimer = null;
let unreadCount = 0;
let latestNextKey = 0;

function updateThemeButton() {
  const isDark = document.body.classList.contains("dark");
  const btn = document.getElementById("themeToggleBtn");
  if (btn) {
    btn.textContent = isDark ? "☀️ Light" : "🌙 Dark";
  }
}

function toggleTheme() {
  const isDark = document.body.classList.toggle("dark");
  try {
    localStorage.setItem("chat_theme", isDark ? "dark" : "light");
  } catch (e) {}
  updateThemeButton();
}

(function initTheme() {
  try {
    if (localStorage.getItem("chat_theme") === "dark") {
      document.body.classList.add("dark");
    }
  } catch (e) {}
})();

function showUnreadBadge(count) {
  if (!unreadBadge || !unreadCountEl) return;
  unreadCountEl.textContent = count;
  unreadBadge.style.display = "flex";
}

function hideUnreadBadge() {
  unreadCount = 0;
  if (unreadBadge) {
    unreadBadge.style.display = "none";
  }
}

function scrollToLatest() {
  window.scrollTo({ top: document.body.scrollHeight, behavior: "smooth" });
  hideUnreadBadge();
}

function scrollTopAndPrint() {
  window.scrollTo(0, 0);
  setTimeout(() => {
    window.print();
  }, 200);
}

function copyPlainText() {
  const text = container.innerText || container.textContent;
  if (navigator.clipboard && window.isSecureContext) {
    navigator.clipboard.writeText(text).then(() => {
      alert("Text copied to clipboard.");
    }).catch(err => {
      alert("Failed to copy: " + err);
    });
  } else {
    const textarea = document.createElement("textarea");
    textarea.value = text;
    textarea.style.position = "fixed";
    document.body.appendChild(textarea);
    textarea.focus();
    textarea.select();
    try {
      const success = document.execCommand("copy");
      alert(success ? "Text copied to clipboard." : "Failed to copy text.");
    } catch (err) {
      alert("Copy failed: " + err);
    }
    document.body.removeChild(textarea);
  }
}

window.addEventListener("scroll", () => {
  const isAtBottom = (window.innerHeight + window.scrollY) >= (document.body.offsetHeight - 50);
  if (isAtBottom && unreadCount > 0) {
    hideUnreadBadge();
  }
});

document.addEventListener("visibilitychange", () => {
  if (document.hidden) {
    if (pollTimer) {
      clearTimeout(pollTimer);
      pollTimer = null;
    }
  } else {
    if (!pollTimer) {
      loadData(latestNextKey);
    }
  }
});

async function loadData(nextMessage) {
  if (pollTimer) {
    clearTimeout(pollTimer);
    pollTimer = null;
  }

  try {
    const params = new URLSearchParams({
      numKey: nextMessage,
      _t: Date.now()
    });

    const response = await fetch("messages?" + params.toString(), { cache: "no-store" });
    if (!response.ok) {
      throw new Error("HTTP error " + response.status);
    }
    const data = await response.json();
    latestNextKey = data.nextKey;

    if (statusEl) {
      statusEl.style.color = "var(--status-color)";
      statusEl.style.animation = "pulse 1.2s infinite ease-in-out";
    }

    if (data.messages && data.messages.length > 0) {
      for (const message of data.messages) {
        const typeStr = message[0];
        const hour = message[1];
        let name = String(message[2]);
        let text = String(message[3]);

        if (text.startsWith("/me ")) {
          text = text.substring(3);
        } else {
          name += ": ";
        }

        const lineDiv = document.createElement("div");
        lineDiv.className = "line";

        const hourSpan = document.createElement("span");
        hourSpan.className = "hour";
        hourSpan.textContent = "[" + hour + "] ";

        const nameSpan = document.createElement("span");
        nameSpan.className = "name-" + typeStr;
        nameSpan.textContent = name;

        const textSpan = document.createElement("span");
        textSpan.className = "text-" + typeStr;
        textSpan.textContent = text;

        lineDiv.appendChild(hourSpan);
        lineDiv.appendChild(nameSpan);
        lineDiv.appendChild(textSpan);

        container.appendChild(lineDiv);
      }

      unreadCount += data.messages.length;
      showUnreadBadge(unreadCount);
    }

    if (!data.done) {
      if (statusEl) {
        statusEl.style.display = "flex";
        statusEl.textContent = "Loading chat (" + container.children.length + " lines)...";
      }
      loadData(latestNextKey);
    } else {
      if (statusEl) {
        statusEl.style.display = "none";
      }
      if (!document.hidden) {
        pollTimer = setTimeout(() => loadData(latestNextKey), POLL_INTERVAL);
      }
    }
  } catch (err) {
    console.error("Fetch error:", err);
    if (statusEl) {
      statusEl.style.display = "flex";
      statusEl.style.animation = "none";
      statusEl.style.color = "#E53935";
      statusEl.textContent = `Connection issue. Retrying in ${Math.round(POLL_INTERVAL / 1000)}s...`;
    }
    if (!document.hidden) {
      pollTimer = setTimeout(() => loadData(nextMessage), POLL_INTERVAL);
    }
  }
}

document.addEventListener("DOMContentLoaded", () => {
  updateThemeButton();
  loadData(0);
});

local url = ""

local function escapeXml(str)
    str = ll.ReplaceSubString(str, "&", "&amp;", 0)
    str = ll.ReplaceSubString(str, "<", "&lt;", 0)
    str = ll.ReplaceSubString(str, ">", "&gt;", 0)
    str = ll.ReplaceSubString(str, '"', "&quot;", 0)
    return str
end

function LLEvents.http_request(id, method, body)
    if method == URL_REQUEST_GRANTED then
        url = body
        print(url .. "/chat")
    elseif method == URL_REQUEST_DENIED then
        print("Unable to get URL!")
    elseif method == "GET" then
        local path = ll.ToLower(ll.GetHTTPHeader(id, "x-path-info"))
        local query = ll.ToLower(ll.GetHTTPHeader(id, "x-query-string"))
        if path == "/chat" then
            local title = escapeXml(ll.GetObjectDesc())
            if title == "" then title = "Chat Log" end
            local page = ll.ReplaceSubString(html, "@TITLE@", title, 0)
            ll.SetContentType(id, CONTENT_TYPE_XHTML)
            ll.HTTPResponse(id, 200, page)
        elseif path == "/messages" then
            local numKey = (tonumber(query:match("numkey=(%d+)")) or 0) + 1
            local totalKeys = ll.LinksetDataCountKeys()
            local messages = {}
            local length = 0
            while numKey <= totalKeys and length < 30000 do
                local lKey = tostring(numKey + 1000)
                local lValue = ll.LinksetDataRead(lKey)
                if #lValue > 0 then
                    table.insert(messages, lljson.decode(lValue))
                    length += #lValue
                end
                numKey += 1
            end
            local payload = {
                messages = messages,
                done = (numKey > totalKeys),
                nextKey = numKey - 1
            }
            ll.SetContentType(id, CONTENT_TYPE_JSON)
            ll.HTTPResponse(id, 200, lljson.encode(payload))
        else
            ll.HTTPResponse(id, 404, "Not Found")
        end
    else
        ll.HTTPResponse(id, 405, "Method Not Allowed")
    end
end

function LLEvents.on_rez(start_param)
    ll.ResetScript()
end

function LLEvents.changed(change)
    if bit32.btest(change, bit32.bor(CHANGED_REGION_START, CHANGED_REGION, CHANGED_TELEPORT, CHANGED_OWNER)) then
        ll.ResetScript()
    end
end
