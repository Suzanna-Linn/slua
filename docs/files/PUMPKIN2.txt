local function timedMessage()
    local randomMessage = math.random(#messages)
    ll.Say(0, messages[randomMessage])
end

function LLEvents.collision_start(events)
    local name = ll.GetDisplayName(events[1]:getKey())
    local randomMessage = math.random(#namedMessages)
    local message = namedMessages[randomMessage]
    message = ll.ReplaceSubString(message, "@NAME@", name, 0)
    ll.Say(0, message)
end

ll.VolumeDetect(true)
LLTimers:every(SECONDS_BETWEEN_MESSAGES, timedMessage)
