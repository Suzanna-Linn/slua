local function show(url)
    ll.SetPrimMediaParams(2, {
        PRIM_MEDIA_CURRENT_URL, url,
        PRIM_MEDIA_HOME_URL, url,
        PRIM_MEDIA_AUTO_ZOOM, false,
        PRIM_MEDIA_FIRST_CLICK_INTERACT, true,
        PRIM_MEDIA_PERMS_INTERACT, PRIM_MEDIA_PERM_ANYONE,
        PRIM_MEDIA_PERMS_CONTROL, PRIM_MEDIA_PERM_NONE,
        PRIM_MEDIA_AUTO_PLAY, true,
        PRIM_MEDIA_WIDTH_PIXELS, 512,
        PRIM_MEDIA_HEIGHT_PIXELS, 512,
    })
end

local function getParam(params, paramName)
    local paramValue = ("&" .. params):match("[?&]" .. paramName .. "=([^&#]*)")
    return if paramValue then ll.UnescapeURL(paramValue:gsub("%+", " ")) else nil
end

local function getSitters()
    local newSitters = {}
    for _, visitor in ll.GetAgentList(AGENT_LIST_REGION, {}) do
        if bit32.btest(ll.GetAgentInfo(visitor), AGENT_SITTING) then
            table.insert(newSitters, visitor)
        end
    end
    table.sort(newSitters, function(a, b) return tostring(a) < tostring(b) end)
    if ll.DumpList2String(sitters, ",") ~= ll.DumpList2String(newSitters, ",") then
        sitters = newSitters
        revision += 1
    end
end

function LLEvents.http_request(id, method, body)
    if method == URL_REQUEST_GRANTED then
        url = body .. "/sitters"
        ll.Say(0, url)
        show(url)
    elseif method == URL_REQUEST_DENIED then
        print("Unable to get URL!")
    elseif method == "GET" then
        local path = ll.ToLower(ll.GetHTTPHeader(id, "x-path-info"))
        local query = ll.ToLower(ll.GetHTTPHeader(id, "x-query-string"))
        if path == "/sitters" then
            ll.SetContentType(id, CONTENT_TYPE_XHTML)
            ll.HTTPResponse(id, 200, htmlPage)
        elseif path == "/sittersdata" then
            local clientRev = getParam(query, "rev") or -1
            if clientRev == revision then
                ll.SetContentType(id, CONTENT_TYPE_JSON)
                ll.HTTPResponse(id, 200, '{"changed":false}')
            else
                getSitters()
                local sittersList = {}
                for _, sitter in sitters do
                    table.insert(sittersList, {
                        name = ll.GetDisplayName(sitter),
                        username = ll.GetUsername(sitter)
                    })
                end
                local payload = {
                    changed = true,
                    rev = revision,
                    sitters = sittersList
                }
                ll.SetContentType(id, CONTENT_TYPE_JSON)
                ll.HTTPResponse(id, 200, lljson.encode(payload))
            end
        else
            ll.HTTPResponse(id, 404, "Not Found")
        end
    end
end

function LLEvents.on_rez(start_param)
    ll.ResetScript()
end

function LLEvents.changed(change)
    if bit32.btest(change, bit32.bor(CHANGED_REGION_START, CHANGED_OWNER, CHANGED_INVENTORY)) then
        ll.ResetScript()
    end
end

ll.RequestURL()
