
local year = 2026
local daysSpecialGift = 20

local top = vector(135, 398, 0)
local bottom = vector(884, 773, 0)
local dayOffset = 2

local notecardConfig = "Configuration"

local CALENDAR_FACE = 2
local ME = ll.GetOwner()

local sounds = {}

local notecardLine = 1
local requestLineId = NULL_KEY
local notecard = {}

local giftsByDay = {}  -- keys 1..24, plus 25 for special gift Dec 25-31
local messages = {}

local size = vector((bottom.x - top.x) / 7, (bottom.y - top.y) / if dayOffset > 4 then 6 else 5, 0)

local ownerMode = "owner" -- "owner", "claim", "visitor", or "day"
local testDay = nil
local pendingReset = false

local setupStep = nil
local setupTop = nil
local setupBottom = nil

local function uuid2byteString(id)
    return llbase64.encode(id.bytes):sub(1, 22)
end

local function byteString2uuid(byteString)
    return uuid(llbase64.decode(byteString .. "==", true))
end

local function getSltDate()
    local diff = ll.GetWallclock() - ll.GetGMTclock()
    if diff > 0 then
        diff -= 86400
    end
    return os.date("!*t", os.time() + math.round(diff / 3600) * 3600)
end

local function getTodayFor(toucher)
    if toucher == ME and ownerMode == "day" and testDay then
        return { year = year, month = 12, day = testDay }
    end

    return getSltDate()
end

local function countDailyGifts(mask)
    local count = 0
    for i = 0, 23 do
        if bit32.btest(mask, bit32.lshift(1, i)) then
            count += 1
        end
    end
    return count
end

local function sendOwnerLines(owner, header, lines)
    if #lines == 0 then
        ll.RegionSayTo(owner, 0, `{header}\n(none)`)
        return
    end

    local chunk = header
    for _, line in lines do
        if #chunk + #line + 1 > 950 then
            ll.RegionSayTo(owner, 0, chunk)
            chunk = line
        else
            chunk ..= "\n" .. line
        end
    end
    ll.RegionSayTo(owner, 0, chunk)
end

local function readVisitors(giftDay, bitMask)
    local visitors = {}
    local totalKeys = ll.LinksetDataCountKeys()
    local start = 1
    while start <= totalKeys do
        local keys = ll.LinksetDataListKeys(start, 100)
        for _, key in keys do
            local data = tonumber(ll.LinksetDataRead(key))
            if giftDay then
                if bit32.btest(data, bitMask) then
                    table.insert(visitors, `secondlife:///app/agent/{byteString2uuid(key)}/about`)
                end
            else
                table.insert(visitors, {
                    id = byteString2uuid(key),
                    days = countDailyGifts(data),
                    hasSpecial = bit32.btest(data, bitMask),
                })
            end
        end
        start += #keys
    end
    return visitors
end

local function ownerDayInfo(owner, day)
    local giftDay = math.min(day, 25)
    local bitMask = bit32.lshift(1, giftDay - 1)
    local lines = readVisitors(giftDay, bitMask)

    local label = if day < 25 then `Day {day}` else "Special Christmas Gift"
    sendOwnerLines(owner, `{label} — {#lines} visitor(s):`, lines)
end

local function ownerSummaryInfo(owner)
    local bitMask = bit32.lshift(1, 24)
    local visitors = readVisitors(nil, bitMask)

    table.sort(visitors, function(a, b)
        if a.days ~= b.days then
            return a.days > b.days
        end
        if a.hasSpecial ~= b.hasSpecial then
            return a.hasSpecial
        end
        return tostring(a.id) < tostring(b.id)
    end)

    local lines = {}
    for _, v in visitors do
        local specialTag = if v.hasSpecial then " (and Special)" else ""
        table.insert(lines, `secondlife:///app/agent/{v.id}/about: {v.days} day(s){specialTag}`)
    end
    sendOwnerLines(owner, `All visitors ({#visitors}), sorted by total days:`, lines)
end

local function giveInfo(toucher, day)
    local monthNames = {
        "January", "February", "March", "April", "May", "June",
        "July", "August", "September", "October", "November", "December",
    }

    if day then
        local today = getTodayFor(toucher)
        local todayStr = `{monthNames[today.month]} {today.day}, {today.year} (SLT)`
        ll.RegionSayTo(toucher, 0, `Wrong day! You touched December {day}, but today is {todayStr}.`)
    end

    ll.RegionSayTo(toucher, 0, `How this Advent Calendar works: Touch today's date from December 1 to 24 ({year}, SLT) to claim that day's gift. Collect at least {daysSpecialGift} daily gifts to unlock the Special Christmas Gift by touching today's date (Dec 25–31) between December 25 and 31!`)
end

local function giveGift(toucher, day)
    local giftDay = math.min(day, 25)
    local bitMask = bit32.lshift(1, giftDay - 1)
    local gifts = giftsByDay[giftDay]

    local strToucher = uuid2byteString(toucher)
    local data = tonumber(ll.LinksetDataRead(strToucher)) or 0

    if giftDay == 25 then
        local days = countDailyGifts(data)
        if days < daysSpecialGift then
            ll.RegionSayTo(toucher, 0, `Sorry, the Special Christmas Gift is only for visitors who have got at least {daysSpecialGift} gifts. You have only collected {days} gift(s).`)
            return
        end
    end

    if not bit32.btest(data, bitMask) then
        data = bit32.bor(data, bitMask)
        ll.LinksetDataWrite(strToucher, tostring(data))

        if #sounds > 0 then
            local sound = sounds[math.random(#sounds)]
            ll.PlaySound(sound, 1)
        end
    else
        ll.RegionSayTo(toucher, 0, "(Already claimed today — re-sending gift)")
    end

    if #messages > 0 then
        local msg = messages[math.random(#messages)]
        ll.RegionSayTo(toucher, 0, msg)
    end

    if not gifts or #gifts == 0 then
        ll.RegionSayTo(toucher, 0, "Sorry, today's gift isn't set up yet!")
        return
    end

    if #gifts == 1 then
        ll.GiveInventory(toucher, gifts[1])
    else
        local folderName = `Advent Calendar - {if day < 25 then "Day " .. day else "Special Christmas Gift"}`
        ll.GiveInventoryList(toucher, folderName, gifts)
    end

    if giftDay < 25 then
        local dailyCount = countDailyGifts(data)
        if dailyCount >= daysSpecialGift then
            ll.RegionSayTo(toucher, 0, `Progress: {dailyCount}/{daysSpecialGift} daily gifts collected — Special Christmas Gift unlocked for Dec 25–31!`)
        else
            ll.RegionSayTo(toucher, 0, `Progress: {dailyCount}/{daysSpecialGift} daily gifts collected toward the Special Christmas Gift!`)
        end
    end
end

local function dayTouched(toucher, touchPos)
    local day = nil
    if touchPos.x >= top.x and touchPos.x < bottom.x and touchPos.y >= top.y and touchPos.y < bottom.y then
        local cell = vector.floor((touchPos - top) / size)
        day = cell.y * 7 + cell.x + 1 - dayOffset
        if day < 1 or day > 31 then
            day = nil
        end
    end

    if toucher == ME and ownerMode == "owner" then
        if day then
            ownerDayInfo(toucher, day)
        else
            ownerSummaryInfo(toucher)
        end
        return
    end

    if day then
        local today = getTodayFor(toucher)
        local forceClaim = (toucher == ME and ownerMode == "claim")
        if forceClaim or (today.year == year and today.month == 12 and today.day == day) then
            giveGift(toucher, day)
            return
        end
    end
    giveInfo(toucher, day)
end

local function checkConfig()
    giftsByDay = {}
    messages = {}
    local section

    for lineNum, rawLine in notecard do
        local line = rawLine:match("^%s*(.-)%s*$")
        if line == "" or line:sub(1, 1) == "#" then
            continue
        end

        local header = line:match("^%[(.-)%]$")
        if header then
            section = header:lower():match("^%s*(.-)%s*$")
            continue
        end

        if section == "settings" then
            local lhs, rhs = line:match("^(.-)%s*[=:]%s*(.+)$")
            if not lhs or not rhs then
                print(`Config warning (line {lineNum}): expected 'Setting = Value', got '{line}'`)
                continue
            end

            local key = lhs:lower():gsub("[%s_%-]", "")
            local v = tonumber(rhs)
            v = if v then v // 1 else nil

            if key == "year" then
                if v and v >= 2000 then 
                    year = v
                else
                    print(`Config warning (line {lineNum}): invalid Year '{rhs}'`)
                end
            elseif key == "daysspecialgift" then
                if v and v >= 0 and v <= 24 then
                    daysSpecialGift = v
                else
                    print(`Config warning (line {lineNum}): invalid Days Special Gift '{rhs}' (use 0-24)`)
                end
            else
                print(`Config warning (line {lineNum}): unknown setting '{lhs}'`)
            end

        elseif section == "gifts" then
            local lhs, rhs = line:match("^(.-)%s*[=:]%s*(.+)$")
            if not lhs or not rhs then
                print(`Config warning (line {lineNum}): expected 'Day = Gift Name', got '{line}'`)
                continue
            end

            local key = lhs:lower():match("^%s*(.-)%s*$")
            local dayNum
            if key == "special" or key == "christmas" then
                dayNum = 25
            else
                dayNum = tonumber(key:match("^day%s*(%d+)$") or key:match("^(%d+)$"))
            end

            if not dayNum or dayNum < 1 or dayNum > 25 then
                print(`Config warning (line {lineNum}): invalid day '{lhs}' (use 1-24 or Special)`)
                continue
            end

            giftsByDay[dayNum] = giftsByDay[dayNum] or {}
            if ll.GetInventoryType(rhs) ~= INVENTORY_NONE then
                table.insert(giftsByDay[dayNum], rhs)
            else
                for _, part in rhs:split(",") do
                    local itemName = part:match("^%s*(.-)%s*$")
                    if itemName ~= "" then
                        if ll.GetInventoryType(itemName) == INVENTORY_NONE then
                            print(`Config warning (line {lineNum}): '{itemName}' not found in Contents`)
                            continue
                        end
                        table.insert(giftsByDay[dayNum], itemName)
                    end
                end
            end

        elseif section == "messages" then
            table.insert(messages, line)

        else
            print(`Config warning (line {lineNum}): put lines under [Settings], [Gifts] or [Messages]`)
        end
    end

    for i = 1, 25 do
        if not giftsByDay[i] or #giftsByDay[i] == 0 then
            print(`Config warning: no gifts for day {i}`)
        end
    end
    if #messages == 0 then
        print("Config warning: no messages found under [Messages]")
    end

    print("Configuration notecard loading and checking finished")
    notecard = {}
end

local function resetTimeOut()
    if pendingReset then
        pendingReset = false
        ll.RegionSayTo(ME, 0, "Reset cancelled (timed out).")
    end
end

local function updateGridSize()
    size = vector((bottom.x - top.x) / 7, (bottom.y - top.y) / if dayOffset > 4 then 6 else 5, 0)
end

local function loadGridFromDesc()
    local desc = ll.GetObjectDesc()
    local tx, ty, bx, by, off = desc:match("^grid:(%-?%d+),(%-?%d+),(%-?%d+),(%-?%d+),(%d+)$")
    if tx then
        local newTop = vector(tonumber(tx), tonumber(ty), 0)
        local newBottom = vector(tonumber(bx), tonumber(by), 0)
        local newOffset = tonumber(off)
        if newBottom.x > newTop.x and newBottom.y > newTop.y and newOffset >= 0 and newOffset <= 6 then
            top = newTop
            bottom = newBottom
            dayOffset = newOffset
            updateGridSize()
        end
    end
end

local function setupTimeOut()
    if setupStep then
        setupStep = nil
        setupTop = nil
        setupBottom = nil
        ll.RegionSayTo(ME, 0, "Calendar setup cancelled (timed out).")
    end
end

local function handleSetupTouch(touchPos)
    LLTimers:off(setupTimeOut)

    if setupStep == 1 then
        setupTop = vector(math.round(touchPos.x), math.round(touchPos.y), 0)
        setupStep = 2
        LLTimers:once(60, setupTimeOut)
        ll.RegionSayTo(ME, 0, `Top-left set to ({setupTop.x}, {setupTop.y}). Setup (2/3): Click the BOTTOM-RIGHT corner of the calendar grid.`)

    elseif setupStep == 2 then
        local candidate = vector(math.round(touchPos.x), math.round(touchPos.y), 0)
        if candidate.x <= setupTop.x or candidate.y <= setupTop.y then
            LLTimers:once(60, setupTimeOut)
            ll.RegionSayTo(ME, 0, "Bottom-right must be to the right and below top-left. Setup (2/3): Click the BOTTOM-RIGHT corner again.")
            return
        end
        setupBottom = candidate
        setupStep = 3
        LLTimers:once(60, setupTimeOut)
        ll.RegionSayTo(ME, 0, `Bottom-right set to ({setupBottom.x}, {setupBottom.y}). Setup (3/3): Click anywhere inside DAY 1.`)

    elseif setupStep == 3 then
        if touchPos.x < setupTop.x or touchPos.x >= setupBottom.x or touchPos.y < setupTop.y or touchPos.y >= setupBottom.y then
            LLTimers:once(60, setupTimeOut)
            ll.RegionSayTo(ME, 0, "Click was outside the grid. Setup (3/3): Click anywhere inside DAY 1.")
            return
        end

        local col = math.clamp(math.floor((touchPos.x - setupTop.x) / ((setupBottom.x - setupTop.x) / 7)), 0, 6)
        local rows = if col > 4 then 6 else 5
        local row = math.floor((touchPos.y - setupTop.y) / ((setupBottom.y - setupTop.y) / rows))
        if row ~= 0 then
            LLTimers:once(60, setupTimeOut)
            ll.RegionSayTo(ME, 0, "Day 1 must be in the top row. Setup (3/3): Click anywhere inside DAY 1.")
            return
        end

        top = setupTop
        bottom = setupBottom
        dayOffset = col
        updateGridSize()

        setupStep = nil
        setupTop = nil
        setupBottom = nil

        ll.SetObjectDesc(`grid:{top.x},{top.y},{bottom.x},{bottom.y},{dayOffset}`)
        ll.RegionSayTo(ME, 0, `Setup complete! Saved to object description: top=({top.x}, {top.y}), bottom=({bottom.x}, {bottom.y}), dayOffset={dayOffset}.`)
    end
end

function LLEvents.touch_start(events)
    for _, ev in events do
        if ev:getTouchFace() == CALENDAR_FACE then
            local toucher = ev:getKey()
            local touchIndex = ev:getTouchST()
            local touchPos = vector(touchIndex.x * 1024, (1 - touchIndex.y) * 1024, 0)
            if toucher == ME and setupStep then
                handleSetupTouch(touchPos)
            else
                dayTouched(toucher, touchPos)
            end
        elseif setupStep and ev:getKey() == ME then
            ll.RegionSayTo(ME, 0, `Please touch face {CALENDAR_FACE} (the calendar face) for setup.`)
        end
    end
end

function LLEvents.dataserver(request, data)
    if request == requestLineId then
        repeat
            if data ~= EOF then
                table.insert(notecard, data)
                notecardLine += 1
                data = ll.GetNotecardLineSync(notecardConfig, notecardLine)
                if data == NAK then
                    requestLineId = ll.GetNotecardLine(notecardConfig, notecardLine)
                end
            end
        until data == EOF or data == NAK
        if data == EOF then
            checkConfig()
        end
    end
end

function LLEvents.changed(change)
    if bit32.btest(change, CHANGED_OWNER) then
        ll.LinksetDataReset()
        ll.ResetScript()
    end
    if bit32.btest(change, CHANGED_INVENTORY) then
        ll.ResetScript()
    end
end

function LLEvents.listen(channel, name, id, message)
    local cmd = message:lower():match("^%s*(.-)%s*$")

    if cmd == "yes" then
        if pendingReset then
            pendingReset = false
            LLTimers:off(resetTimeOut)
            ll.LinksetDataReset()
            ll.RegionSayTo(ME, 0, "Cleared ALL LinksetData.")
        else
            ll.RegionSayTo(ME, 0, "No reset is pending.")
        end
        return
    end

    if pendingReset then
        pendingReset = false
        LLTimers:off(resetTimeOut)
        if cmd ~= "reset" then
            ll.RegionSayTo(ME, 0, "Pending reset cancelled.")
        end
    end

    if setupStep then
        setupStep = nil
        setupTop = nil
        setupBottom = nil
        LLTimers:off(setupTimeOut)
        if cmd ~= "setup" then
            ll.RegionSayTo(ME, 0, "Calendar setup cancelled.")
        end
    end

    if cmd == "owner" or cmd == "claim" or cmd == "visitor" then
        ownerMode = cmd
        testDay = nil
        ll.RegionSayTo(ME, 0, `Owner mode set to: {ownerMode}`)

    elseif cmd == "day" or cmd:match("^day%s+") then
        local dayNum = tonumber(cmd:match("^day%s+(%d+)$"))
        if dayNum and dayNum >= 1 and dayNum <= 31 then
            ownerMode = "day"
            testDay = dayNum
            ll.RegionSayTo(ME, 0, `Owner mode set to: simulated December {testDay}, {year} (SLT)`)
        else
            ll.RegionSayTo(ME, 0, "Usage: /99 day <1-31>")
        end

    elseif cmd == "clear" then
        ll.LinksetDataDelete(uuid2byteString(ME))
        ll.RegionSayTo(ME, 0, "Cleared your own LinksetData test progress.")

    elseif cmd == "reset" then
        pendingReset = true
        LLTimers:once(15, resetTimeOut)
        ll.RegionSayTo(ME, 0, "WARNING: This will erase ALL visitors' LinksetData! Type '/99 yes' within 15 seconds to confirm.")

    elseif cmd == "setup" then
        setupStep = 1
        setupTop = nil
        setupBottom = nil
        LLTimers:once(60, setupTimeOut)
        ll.RegionSayTo(ME, 0, "Calendar setup (1/3): Click the TOP-LEFT corner of the calendar grid.")

    else
        ll.RegionSayTo(ME, 0, "Commands: /99 owner | /99 claim | /99 visitor | /99 day <1-31> | /99 clear | /99 reset | /99 setup")
    end
end

ll.Listen(99, "", ME, "")

loadGridFromDesc()

for i = 1, ll.GetInventoryNumber(INVENTORY_SOUND) do
    table.insert(sounds, ll.GetInventoryName(INVENTORY_SOUND, i))
end

if ll.GetInventoryType(notecardConfig) == INVENTORY_NOTECARD then
    requestLineId = ll.GetNotecardLine(notecardConfig, notecardLine)
else
    print(`Configuration notecard missing: {notecardConfig}`)
end
