local CHANNEL_TIPJAR = 15
local PUBLIC_THANKS = true

local payButtons = {25, 50, 100, 250}  -- = { 150, PAY_HIDE, PAY_HIDE, PAY_HIDE}
local payPrice = PAY_DEFAULT  -- = PAY_HIDE
local myName = ll.GetDisplayName(ll.GetOwner())
local lastAmount = 0
local tipjarAmount = 0
local totalAmount = 0
local debitPerms = false

local function showText()
    local text = ""
    text = myName .. "'s tipjar"
    text ..= "\n"
    text ..= "\nTotal Amount: " .. totalAmount
    text ..= "\nTipjar tips: " .. tipjarAmount
    text ..= "\nLast tip: " .. lastAmount
    text ..= "\n \n \n"
    ll.SetText(text, colors.WHITE, 1)
end

local function sayThanks(tipperId)
    local thanksMessage = "Thank you very much, " .. ll.GetDisplayName(tipperId) .. " for your splendorous tip!"
    if PUBLIC_THANKS then
        local tipperPos = ll.GetObjectDetails(tipperId, {OBJECT_POS})[1]
        local tipjarPos = ll.GetPos()
        local tipperDistance = ll.VecDist(tipjarPos, tipperPos)
        if tipperDistance < 20 then
            ll.Say(0, thanksMessage)
        elseif tipperDistance < 100 then
            ll.Shout(0, thanksMessage)
        else
            ll.Shout(0, thanksMessage)
            ll.RegionSayTo(tipperId, 0, thanksMessage)
        end
    else
        ll.RegionSayTo(tipperId, 0, thanksMessage)
    end
end

function LLEvents.run_time_permissions(perm)
    if bit32.btest(perm, PERMISSION_DEBIT) then
        debitPerms = true
    end
end

function LLEvents.money(id, amount)
    local SHARE_WITH_ID = uuid("")  -- add the uuid of the person to share
    local SHARE_PER_CENT = 50
    local shareAmount = 0
    shareAmount = math.round(amount * SHARE_PER_CENT // 100)
    if SHARE_WITH_ID.istruthy then
        if debitPerms then
            if shareAmount <= amount then  -- Assert that it doesn't give more that tipped
                ll.GiveMoney(SHARE_WITH_ID, shareAmount)
            else
                ll.OwnerSay("It was going to give more than tipped!!!")
            end
        end
    end
    sayThanks(id)
    lastAmount = amount
    tipjarAmount += amount
    totalAmount += amount
    showText()
    ll.RegionSay(CHANNEL_TIPJAR, tostring(amount))
end

function LLEvents.listen(channel, name, id, message)
    if channel == CHANNEL_TIPJAR then
        totalAmount += tonumber(message)
        showText()
    end

end

function LLEvents.on_rez(start_param)
    ll.ResetScript()
end

function LLEvents.changed(change)
    if bit32.btest(change, bit32.bor(CHANGED_OWNER, CHANGED_INVENTORY)) then
        ll.ResetScript()
    end
end

debitPerms = false
ll.RequestPermissions(ll.GetOwner(), PERMISSION_DEBIT)
ll.SetPayPrice(PAY_DEFAULT, payButtons)
ll.SetClickAction(CLICK_ACTION_PAY)
showText()
ll.Listen(CHANNEL_TIPJAR, "", "", "")
