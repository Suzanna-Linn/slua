local MIN_VERTEX = vector(0, 0, 0)
local MAX_VERTEX = vector(256, 256, 4096)
local MAX_DISTANCE = 4112

local VERTEX_LOW = vector(9, 6, 602)
local VERTEX_HIGH = vector(53, 56, 622)
local DISTANCE = MAX_DISTANCE
local AGENT_SELECTION = AGENT_LIST_PARCEL

local SCAN_SECONDS = 15

local WEEKDAY_NAMES = {"Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"}
local WEEKDAY_ORDER = {2, 3, 4, 5, 6, 7, 1} -- Mon to Sun

local PERIOD_LABELS = {
    {key = "today", label = "TODAY", rowClass = "row-today"},
    {key = "yesterday", label = "YESTERDAY", rowClass = ""},
    {key = "thisMonth", label = "THIS MONTH", rowClass = ""},
    {key = "lastMonth", label = "LAST MONTH", rowClass = ""},
    {key = "allTime", label = "ALL TIME", rowClass = "row-alltime"},
}

local checkBounds = VERTEX_LOW ~= MIN_VERTEX or VERTEX_HIGH ~= MAX_VERTEX
local checkDistance = DISTANCE ~= MAX_DISTANCE
local checkInPlace = checkBounds or checkDistance

local people = {}
local totalPeople = 0
local todayVisitors = {}

local function getSltOffset()
    local diff = ll.GetWallclock() - ll.GetGMTclock()
    if diff > 0 then
        diff -= 86400
    end
    return math.round(diff / 3600) * 3600
end

local lastTimestamp = os.time()
local todayStartTimestamp = lastTimestamp
local lastSltOffset = getSltOffset()
local lastSltDay = (lastTimestamp + lastSltOffset) // 86400
local lastSltDate = os.date("!*t", lastTimestamp + lastSltOffset)

local function initStat()
    local hourlyTime = {}
    for hour = 0, 23 do
        hourlyTime[hour] = 0
    end
    return {
        totalVisits = 0,
        uniqueVisitors = 0,
        totalTime = 0,
        days = 0,
        hourlyTime = hourlyTime,
    }
end

local stats = {
    today = initStat(),
    yesterday = initStat(),
    thisMonth = initStat(),
    lastMonth = initStat(),
    allTime = initStat(),
}

local allTimeWeekdays = {}
for wday = 1, 7 do
    allTimeWeekdays[wday] = {totalTime = 0, days = 0}
end

local function addTime(fromTs, toTs, sltOffset, count)
    if toTs <= fromTs or count <= 0 then
        return
    end
    stats.today.totalTime += (toTs - fromTs) * count
    local cursor = fromTs
    while cursor < toTs do
        local sltHourIndex = (cursor + sltOffset) // 3600
        local hour = sltHourIndex % 24
        local nextHourUtc = (sltHourIndex + 1) * 3600 - sltOffset
        local sliceEnd = math.min(toTs, nextHourUtc)
        stats.today.hourlyTime[hour] += (sliceEnd - cursor) * count
        cursor = sliceEnd
    end
end

local function newDay(nowTimestamp, nowSltOffset, nowSltDay, nowSltDate)
    local endOfLastDayUtc = (lastSltDay + 1) * 86400 - lastSltOffset
    addTime(lastTimestamp, endOfLastDayUtc, lastSltOffset, totalPeople)
    local wday = lastSltDate.wday
    allTimeWeekdays[wday].totalTime += stats.today.totalTime
    allTimeWeekdays[wday].days += 1
    stats.yesterday = stats.today
    for _, info in {"totalVisits", "uniqueVisitors", "totalTime", "days"} do
        stats.thisMonth[info] += stats.today[info]
        stats.allTime[info] += stats.today[info]
    end
    for hour = 0, 23 do
        stats.thisMonth.hourlyTime[hour] += stats.today.hourlyTime[hour]
        stats.allTime.hourlyTime[hour] += stats.today.hourlyTime[hour]
    end
    if nowSltDate.month ~= lastSltDate.month or nowSltDate.year ~= lastSltDate.year then
        stats.lastMonth = stats.thisMonth
        stats.thisMonth = initStat()
    end
    stats.today = initStat()
    stats.today.days = 1
    todayVisitors = {}
    people, totalPeople = {}, 0
    lastSltOffset = nowSltOffset
    lastSltDay = nowSltDay
    lastSltDate = nowSltDate
    local startOfTodayUtc = nowSltDay * 86400 - nowSltOffset
    if nowTimestamp - startOfTodayUtc <= SCAN_SECONDS * 2 then
        lastTimestamp = startOfTodayUtc
    else
        lastTimestamp = nowTimestamp
    end
    todayStartTimestamp = lastTimestamp
end

local function personInPlace(personId, myPos)
    local pos = ll.GetObjectDetails(personId, {OBJECT_POS})[1]
    if not pos then
        return false
    end
    if checkBounds and vector.clamp(pos, VERTEX_LOW, VERTEX_HIGH) ~= pos then
        return false
    end
    if checkDistance and ll.VecDist(myPos, pos) > DISTANCE then
        return false
    end
    return true
end

local function scanPeople()
    local nowTimestamp = os.time()
    local nowSltOffset = getSltOffset()
    local nowSltDay = (nowTimestamp + nowSltOffset) // 86400
    if nowSltDay > lastSltDay then
        local nowSltDate = os.date("!*t", nowTimestamp + nowSltOffset)
        newDay(nowTimestamp, nowSltOffset, nowSltDay, nowSltDate)
    else
        lastSltOffset = nowSltOffset
    end
    local myPos = if checkDistance then ll.GetPos() else nil
    local peopleRegion = ll.GetAgentList(AGENT_SELECTION, {})
    local peopleNow = {}
    totalPeople = 0
    for _, personId in peopleRegion do
        if not checkInPlace or personInPlace(personId, myPos) then
            peopleNow[personId] = true
            totalPeople += 1
            if not people[personId] then
                stats.today.totalVisits += 1
                if not todayVisitors[personId] then
                    stats.today.uniqueVisitors += 1
                    todayVisitors[personId] = true
                end
            end
        end
    end
    people = peopleNow
    addTime(lastTimestamp, nowTimestamp, lastSltOffset, totalPeople)
    lastTimestamp = nowTimestamp
end

local function renderVisitorsHtml()
    local rows = {}
    for _, item in PERIOD_LABELS do
        local p = stats[item.key]
        local total = p.totalVisits
        local unique = p.uniqueVisitors
        local seconds = p.totalTime
        local days = p.days
        local visitsPerDay = 0
        local uniquePerDay = 0
        local avgVisitors = 0
        local avgMinutes = 0
        if total > 0 and unique > 0 and days > 0 then
            visitsPerDay = total / days
            uniquePerDay = unique / days
            if item.key == "today" then
                local elapsed = lastTimestamp - todayStartTimestamp
                avgVisitors = if elapsed > 0 then seconds / days / elapsed else 0
            else
                avgVisitors = seconds / days / 86400
            end
            avgMinutes = seconds / unique / 60
        end
        table.insert(rows, string.format(
            "<tr class='%s'>" ..
            "<td class='period'>%s</td>" ..
            "<td class='num-visits'>%.0f</td>" ..
            "<td class='num-unique'>%.0f</td>" ..
            "<td class='num-avg'>%.2f</td>" ..
            "<td class='num-min last'>%.2f</td>" ..
            "</tr>",
            item.rowClass,
            item.label,
            visitsPerDay,
            uniquePerDay,
            avgVisitors,
            avgMinutes
        ))
    end
    local allDays = stats.allTime.days
    local hourlyAvgs = {}
    local maxHourly = 0
    for hour = 0, 23 do
        local avg = if allDays > 0 then stats.allTime.hourlyTime[hour] / allDays / 3600 else 0
        hourlyAvgs[hour] = avg
        if avg > maxHourly then
            maxHourly = avg
        end
    end
    local hourlyBars = {}
    for hour = 0, 23 do
        local avg = hourlyAvgs[hour]
        local pct = if maxHourly > 0 then math.round((avg / maxHourly) * 100) else 0
        if avg > 0 and pct < 4 then
            pct = 4
        end
        local isPeak = maxHourly > 0 and avg == maxHourly
        local colClass = if isPeak then "h-col peak" else "h-col"
        table.insert(hourlyBars, string.format(
            "<div class='%s'>" ..
            "<div class='h-val'>%.1f</div>" ..
            "<div class='h-track'><div class='h-fill' style='height:%d%%'></div></div>" ..
            "<div class='h-lbl'>%02d</div>" ..
            "</div>",
            colClass,
            avg,
            pct,
            hour
        ))
    end
    local weekdayAvgs = {}
    local maxWeekday = 0
    for _, wday in WEEKDAY_ORDER do
        local wd = allTimeWeekdays[wday]
        local avg = if wd.days > 0 then wd.totalTime / wd.days / 86400 else 0
        weekdayAvgs[wday] = avg
        if avg > maxWeekday then
            maxWeekday = avg
        end
    end
    local weekdayCards = {}
    for _, wday in WEEKDAY_ORDER do
        local wd = allTimeWeekdays[wday]
        local avg = weekdayAvgs[wday]
        local pct = if maxWeekday > 0 then math.round((avg / maxWeekday) * 100) else 0
        if avg > 0 and pct < 5 then
            pct = 5
        end
        local isPeak = maxWeekday > 0 and avg == maxWeekday
        local cardClass = if isPeak then "w-card peak" else "w-card"
        table.insert(weekdayCards, string.format(
            "<div class='%s'>" ..
            "<div class='w-name'>%s</div>" ..
            "<div class='w-val'>%.2f</div>" ..
            "<div class='w-track'><div class='w-fill' style='width:%d%%'></div></div>" ..
            "<div class='w-days'>%d d</div>" ..
            "</div>",
            cardClass,
            WEEKDAY_NAMES[wday],
            avg,
            pct,
            wd.days
        ))
    end
    local body = htmlVisitors
    body = ll.ReplaceSubString(body, "@LIVE_COUNT@", tostring(totalPeople), 0)
    body = ll.ReplaceSubString(body, "@SLT_TIME@", os.date("!%Y-%m-%d %H:%M SLT", lastTimestamp + lastSltOffset), 0)
    body = ll.ReplaceSubString(body, "@TABLE_ROWS@", table.concat(rows), 0)
    body = ll.ReplaceSubString(body, "@HOURLY_BARS@", table.concat(hourlyBars), 0)
    body = ll.ReplaceSubString(body, "@WEEKDAY_CARDS@", table.concat(weekdayCards), 0)
    local html = ll.ReplaceSubString(htmlHeader, "@STYLE@", htmlStyle, 0)
    html = ll.ReplaceSubString(html, "@BODY@", body, 0)
    return html
end

function LLEvents.http_request(id, method, body)
    if method == URL_REQUEST_GRANTED then
        url = body .. "/visitors"
        ll.Say(0, url)
        show(url)
    elseif method == URL_REQUEST_DENIED then
        print("Unable to get URL!")
    elseif method == "GET" then
        local path = ll.ToLower(ll.GetHTTPHeader(id, "x-path-info"))
        if path == "/visitors" or path == "" or path == "/" then
            ll.SetContentType(id, CONTENT_TYPE_XHTML)
            ll.HTTPResponse(id, 200, renderVisitorsHtml())
        else
            ll.HTTPResponse(id, 404, "Not found")
        end
    end
end
